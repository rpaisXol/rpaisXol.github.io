# 7.1 텐서플로 임베딩

- 신경망을 훈련하여 학습된 단어의 벡터 표현
- Dense와 Conv2D처럼 tf.keras는 층을 사용해 임베딩을 구현
- 이 층은 정수를 임베딩 벡터로 매핑하는 룩업 테이블을 만듬
- 이 테이블은 정수로 표현된 단어의 벡터를 담고 있다.
- 실제 자연어 처리 문제에서는 2차원보다 훨씬 더 많은 차원을 사용 
- 벡터 공간에 있는 한 벡터의 방향이 단어의 의미를 인코딩
- 즉, 거의 비슷한 방향을 가리키는 벡터를 가진 단어는 서로 연관

```python

import json
import string
from bs4 import BeautifulSoup

# 1. 문장, 레이블, URL을 저장할 리스트
sentences = []
labels = []
urls = []

# 2. 구두점 제거를 위한 변환 테이블
# string.punctuation: 일반적인 영어 구두점 목록
# translate(): 지정된 구두점 제거
table = str.maketrans('', '', string.punctuation)

# 3. 앞에서 정의한 불용어 리스트 사용
# set으로 변환하여 검색 속도 향상
stopword_set = set(word.lower() for word in stopwords)

# 4. Sarcasm 데이터셋 불러오기
# JSONL 형식: 한 줄에 하나의 JSON 객체 저장
# utf-8-sig: UTF-8 BOM이 있는 경우에도 처리
with open("Sarcasm.json", 'r', encoding='utf-8-sig') as f:

    # 5. JSON 파일을 한 줄씩 읽기
    for line in f:

        # 빈 줄 건너뛰기
        if not line.strip():
            continue

        # JSON 문자열을 Python 딕셔너리로 변환
        item = json.loads(line)

        # 6. 뉴스 헤드라인을 소문자로 변환
        sentence = item['headline'].lower()

        # 7. 구두점 앞뒤에 공백 추가
        sentence = sentence.replace(",", " , ")
        sentence = sentence.replace(".", " . ")
        sentence = sentence.replace("_", " _ ")
        sentence = sentence.replace("/", " / ")

        # 8. BeautifulSoup로 HTML 태그 제거
        soup = BeautifulSoup(sentence, 'html.parser')
        sentence = soup.get_text(' ')

        # 9. 공백을 기준으로 단어 분리
        words = sentence.split()

        # 10. 불용어 제거
        filtered_sentence = ""

        for word in words:

            # 구두점 제거
            word = word.translate(table)

            # 빈 문자열과 불용어 제외
            if word and word not in stopword_set:
                filtered_sentence += word + " "

        # 11. 전처리된 문장 저장
        sentences.append(filtered_sentence.strip())

        # 12. 풍자 여부 레이블 저장
        # 0: 일반 뉴스, 1: 풍자 뉴스
        labels.append(int(item['is_sarcastic']))

        # 13. 기사 URL 저장
        urls.append(item['article_link'])

# 14. 결과 확인
print("전체 문장 수:", len(sentences))
print("전체 레이블 수:", len(labels))
print("전체 URL 수:", len(urls))

print("첫 번째 문장:", sentences[0])
print("첫 번째 레이블:", labels[0])
print("첫 번째 URL:", urls[0])
```

```python
# 2) Tokenizer 및 패딩: 텍스트를 동일한 길의 정수 시퀸스로 변환

# 1. 하이퍼파라미터 설정
vocab_size = 10000       # 최대 어휘 크기
embedding_dim = 16       # 임베딩 벡터 차원
max_length = 10          # 최대 문장 길이
trunc_type = 'post'      # 긴 문장은 뒤쪽 토큰 제거
padding_type = 'post'    # 짧은 문장은 뒤쪽에 0 추가
oov_tok = '<OOV>'        # 미등록 단어를 대체할 토큰

# 2. Tokenizer 생성
tokenizer = Tokenizer(
    num_words=vocab_size,
    oov_token=oov_tok
)

# 3. 학습 문장으로 단어 사전 생성
tokenizer.fit_on_texts(training_sentences)

# 4. 학습 문장을 정수 시퀀스로 변환
training_sequences = tokenizer.texts_to_sequences(
    training_sentences
)

# 5. 학습 문장 패딩
training_padded = pad_sequences(
    training_sequences,
    maxlen=max_length,
    padding=padding_type,
    truncating=trunc_type
)

# 6. 테스트 문장을 정수 시퀀스로 변환
testing_sequences = tokenizer.texts_to_sequences(
    testing_sentences
)

# 7. 테스트 문장 패딩
testing_padded = pad_sequences(
    testing_sequences,
    maxlen=max_length,
    padding=padding_type,
    truncating=trunc_type
)

# 8. NumPy 배열 변환
training_padded = np.array(training_padded, dtype=np.int32)
training_labels = np.array(training_labels, dtype=np.int32)

testing_padded = np.array(testing_padded, dtype=np.int32)
testing_labels = np.array(testing_labels, dtype=np.int32)

# 9. 결과 확인
print("학습 데이터:", training_padded.shape)
print("테스트 데이터:", testing_padded.shape)
print("단어 사전 크기:", len(tokenizer.word_index))
```

```python
# 3) 신경망 정의: 임베딩 기반 이진 분류 모델 구성

# 1. 임베딩 층을 초기화하기 위해
# 어휘 사전 크기와 임베딩 차원을 먼저 지정
vocab_size = 10000
embedding_dim = 16

# 2. 신경망 모델 구성
model = Sequential([

    # 입력층: 정수로 변환된 10개의 토큰 입력
    Input(shape=(max_length,)),

    # Embedding: 각 단어를 16차원 벡터로 변환
    # mask_zero=True: 패딩값 0을 마스킹
    Embedding(
        input_dim=vocab_size,
        output_dim=embedding_dim,
        mask_zero=True
    ),

    # 단어 임베딩 벡터들의 평균 계산
    # 문장의 특징을 하나의 16차원 벡터로 압축
    GlobalAveragePooling1D(),

    # 은닉층: 뉴런 24개, ReLU 활성화 함수
    Dense(24, activation='relu'),

    # 출력층: 풍자 여부를 0~1 사이의 확률로 예측
    Dense(1, activation='sigmoid')
])

# 3. 모델 컴파일
model.compile(
    loss='binary_crossentropy',  # 이진 분류 손실 함수
    optimizer='adam',            # Adam 옵티마이저
    metrics=['accuracy']         # 분류 정확도
)

# 4. 모델 구조 확인
model.summary()
```

1. 임베딩 층을 초기화하기 위해서 어휘 사전 크기와 임베딩 차원을 먼저 지정
2. 각 단어에 대해 embedding_dim 차원의 배열을 초기화, 예를 들면 embedding_dim이 16이라면 어휘 사전에 있는 각 단어는 16차원 벡터에 할당
3. 훈련 데이터의 입력과 레이블을 매핑하기 위해 신경망 훈련될 때 역전파를 통해서 이 벡터가 학습함

- 임베딩 층의 어휘 사전 크기는 10,000이고 임베딩 차원(각 단어의 벡터)은 16
- 임베딩 층의 훈련 가능한 전체 파라미터 개수는 160,000개 이상
- 평균 풀링 층은 임베딩 층의 출력을 단순히 평균해 16차원의 벡터를 만들기 위해 훈련 가능한 파라미터가 0개
- 24개의 뉴런을 가진 밀집 층에 이를 전달하고, 밀집 층은 가중치와 절편을 가지고 있으므로 (24 X 16) + 16 = 408개의 파라미터를 학습
- 이 층의 출력이 하나의 뉴런을 가진 마지막 층에 전달되고, 이 층은 (1 X 24) + 1 = 25개의 파라미터를 가짐
- 모델 훈련 시 30번의 epochs 후 99% 이상 높은 정확도를 얻음, 그러나 검증 정확도는 81%

```python

# epochs=30: 최대 30회 학습
# batch_size=32: 한 번에 32개 데이터 처리

    validation_split=0.2,
```

## 7.1.2 언어 모델의 과대 적합 줄이기

- 신경망이 훈련 데이터에서 과도하게 맞춰질 경우는 과대적합 현상 발생
- 훈련 데이터의 다른 데이터보다 잡은 데이터에 있는 패턴에 잘 맞춰질 경우가 많이 발생
- 이런 특별한 잡음이 검증 세트에는 없기 때문에 신경망이 이를 잘 마출 수 있도록 검증 세트의 손실이 더 나빠짐

### 해결방법 1: 학습률 조정하기

- 과대적합을 일으키는 가장 큰 요소는 옵티마이저의 학습률이 너무 높은 경우
- 즉, 신경망이 너무 빠르게 학습한다는 의미
- 예를 들면 옵티마이저 함수를 기존 adam에서 매개 변수를 사용해 변경 가능

```python
from tensorflow.keras.optimizers import Adam

adam = Adam(
    learning_rate=0.0001,  # 학습률: 가중치 업데이트 크기 조절
    beta_1=0.9,            # 기울기 1차 모멘트의 지수이동평균 계수
    beta_2=0.999,          # 기울기 제곱의 2차 모멘트 지수이동평균 계수
    amsgrad=False          # AMSGrad 변형 알고리즘 사용 여부
)

# 2. 모델 컴파일
model.compile(
    loss='binary_crossentropy',  # 이진 분류 손실 함수
    optimizer=adam,              # 직접 설정한 Adam 옵티마이저 사용
```

### 패딩(padding)

- 신경망 훈련 시 일반적으론 모든 데이터는 크기가 동일 해야 함
- 이미지로 신경망 훈련 시 이미지는 동일한 너비와 높이로 바꿨음
- 텍스트에서도 동일한 이슈가 발생, 단어를 토큰화하고 문장을 토큰의 시퀸스로 바꾸면 시퀸스마타 길이가 다름
- 이때 동일한 길이로 맞추기 위해 패딩(padding)을 사용

```python
# 문장 생성 코드
sentences = [
    'Today is a sunny day',
    'Today is a rainy day',
    'Is it sunny today?',
    'I really enjoyed walked in the snow today'
]

tokenizer.fit_on_texts(sentences)
sequences = tokenizer.texts_to_sequences(sentences)
print(sequences)
```

```text
[[2, 3, 4, 6, 5], 
[2, 3, 4, 7, 5], 
[3, 8, 6, 2], 
[9, 10, 11, 12, 13, 14, 15, 2]]
```

동일한 길이로 만드려면 pad_sequences 함수를 사용할 수 있다.

```python
from tensorflow.keras.preprocessing.sequence import pad_sequences

padded = pad_sequences(sequences)
print(padded)
```

```text
[[ 0  0  0  2  3  4  6  5]
 [ 0  0  0  2  3  4  7  5]
 [ 0  0  0  0  3  8  6  2]
 [ 9 10 11 12 13 14 15  2]]
```

- 보기 좋게 한 줄씩, 시퀸스 길이도 맞춰진 상태로 결과가 출력
- 시퀸스는 단어 리스트에 없는 토큰인 0으로 패딩
- 짧은 문장을 긴 문장의 길이에 맞추기 위해 시작 부분을 0으로 패딩
- 이를 프리패딩(prepadding)이라고 부르며, 이 함수의 기본값
- padding 매개변수에서 이 값을 바꿀 수 있음.

```python
# 프리패딩 적용
padded = pad_sequences(sequences, padding='post')
print(padded)
```

```text
[[ 2  3  4  6  5  0  0  0]
 [ 2  3  4  7  5  0  0  0]
 [ 3  8  6  2  0  0  0  0]
 [ 9 10 11 12 13 14 15  2]]
```

- 패딩된 시퀸스 앞에 단어가 나오고 0이 끝에 추가
- 기본적으로 모든 문장은 가장 긴 문장의 길이에 맞춰 패딩
- 이렇게 하면 잃는 데이터가 없으므로 매우 합리적
- 그 대신에 패딩이 많아짐
- 아주 긴 문장 하나에 맞추기 위해서 패딩이 너무 많이 추가되는 것을 원하지 않는다면?
- 이런 경우는 pad_seqences를 호출할 때 maxlen 매개변수에 최대 길이 지정

```python
# 패딩 길이 설정
padded = pad_sequences(sequences, padding='post', maxlen=6)
print(padded)
```

```text
[[ 2  3  4  6  5  0]
 [ 2  3  4  7  5  0]
 [ 3  8  6  2  0  0]
 [11 12 13 14 15  2]]
```

- 그러나 가장 긴 문장의 단어 일부 손실, 문장의 시작 부분도 잘렸음
- 시작 부분이 아니라 문장 끝 부분의 단어를 자르려면 truncating 매개변수를 사용해 바꿀 수 있음

```python
# 문장 끝 부분 잘림
padded = pad_sequences(sequences, padding='post', maxlen=6, truncating='post')
print(padded)
```

# 6.2 불용어 제거와 텍스트 정제

- 실전 데이터셋은 원치 않는 텍스트가 들어 있는 경우가 대부분임
- the, and, but과 같이 너무 자주 등장해 특별한 의미가 없는 불용어(stopword)를 제거
- 텍스트에 들어 있는 html 테그를 제거하고 싶은 경우도 있다.
- 무례한 단어, 고유명사, 구두점 등도 제거하고 싶은 경우도 있다.

불용어를 제거하는 방법은 불용어 리스트를 준비하고 문장에서 이를 제거, 전처리하는 방식

- 또 다른 작업은 불용어 삭제를 방해하는 구두점 제거 
- 앞의 코드는 공백으로 둘러쌓인 불용어만 찾기 때문에 마침표나 쉼표 앞에 오는 불용어를 찾지 못함
- 파이썬 string 패키지에서 제공하는 translate 함수를 사용하면 이런 문제를 해결 가능
- 또한 string.punctuation으로 일반적인 구두점 목록을 제공

```python

# 구두점 목록 확인
# string.punctuation: 일반적인 ASCII 구두점 제공

sentences = [
    'Today is a sunny day.',
    'Today is a rainy day!',
    'This is a beautiful morning, and I am happy.'
]

print("구두점 목록:", string.punctuation)

# 구두점 제거를 위한 변환 테이블 생성
# str.maketrans(변환 전, 변환 후, 삭제할 문자)
# 세 번째 인자에 구두점을 전달하면 해당 문자를 삭제
table = str.maketrans('', '', string.punctuation)

# 전처리 결과 저장 리스트
filtered_sentences = []

# 문장별 구두점 및 불용어 제거
for sentence in sentences:

    # 구두점 제거
    sentence = sentence.translate(table)

    # 공백을 기준으로 단어 분리
    words = sentence.split()

    # 불용어를 제거한 단어 저장
    filtered_words = []

    for word in words:

        # 소문자로 변환한 후 불용어 여부 확인
        if word.lower() not in stopwords:
            filtered_words.append(word)

    # 남은 단어를 공백으로 연결
    filtered_sentence = ' '.join(filtered_words)

    # 전처리 결과 저장
    filtered_sentences.append(filtered_sentence)

# 결과 출력
print("전처리 결과:", filtered_sentences)
```

```python
        # 빈 문자열과 불용어는 제외
        # 불용어에 포함되지 않은 단어만 추가
        if word and word not in stopword_set:
            filtered_sentence = filtered_sentence + word + " "

    # 7. 전처리된 문장을 리스트에 추가
    # strip(): 문장 앞뒤의 불필요한 공백 제거
    imdb_sentences.append(filtered_sentence.strip())


# 8. Tokenizer 생성
# num_words=25000: 사용할 단어 인덱스 범위를 제한
tokenizer = tf.keras.preprocessing.text.Tokenizer(
    num_words=25000
)

# 9. 전처리된 문장으로 단어 사전 생성
# 각 단어의 등장 빈도를 계산하고 정수 인덱스를 부여
tokenizer.fit_on_texts(imdb_sentences)

# 10. 문장을 정수 시퀀스로 변환
# 각 단어를 단어 사전에 등록된 정수 인덱스로 변환
sequences = tokenizer.texts_to_sequences(imdb_sentences)

# 11. 결과 확인
```

- 오픈소스 데이터셋인 Sentiment Analysis in Text
- 두 개의 데이터 셋 사용, 하나는 이진 분류를 위해 감성을 긍정과 부정으로 축소한 데이터셋이며 다른 하나는 전체 레이블에서 사용하는 데이터 셋,
- 두 데이터셋의 구조는 동일하므로 이진 버전을 사용할 것
- 파이썬 csv 라이브러리 이용, 사용할 데이터는 한 줄에 두 개의 값을 담고 있다.
- 첫 번째 값은 긍정인지 부정인지 나타내는 숫자(0, 1), 두 번째 값은 텍스트 문자열

```python
import csv

        sentences.append(filtered_sentence.strip())
```

## 6.3.3 JSON 파일에서 텍스트 읽기

- 자주 사용하는 또 다른 텍스트 파일 포맷은 JSON이다. 
- 특히 웹 애플리케이션에서 데이터 교환을 위한 많이 사용되는 공개 표준 파일 포맷
- 사람이 알이보기 쉽게 이름/값 쌍을 사용하도록 고안

### JSON 파일 읽기

- 파이썬 json 라이브러리를 사용하면 json 파일을 쉽게 읽음
- 이름/값 쌍을 사용하는 JSON 이름으로 내용을 참조

```python


import json
import string
import tensorflow as tf
from bs4 import BeautifulSoup
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences

# 1. 문장, 레이블, URL을 저장할 리스트 생성
sentences = []   # 전처리된 뉴스 헤드라인
labels = []      # 풍자 여부 (0: 일반 뉴스, 1: 풍자 뉴스)
urls = []        # 뉴스 기사 URL

# 2. 구두점 제거를 위한 변환 테이블 생성
# string.punctuation: 일반적인 영어 구두점 목록
# translate(): 구두점을 공백으로 변환하여 단어 경계 유지
table = str.maketrans({
    char: ' ' for char in string.punctuation
})

# 3. 앞에서 정의한 불용어 리스트 사용
# 불용어 검색 속도를 높이기 위해 set으로 변환
stopword_set = set(word.lower() for word in stopwords)

# 부정어는 문장의 의미에 중요하므로 제거 대상에서 제외
stopword_set -= {'not', 'no', 'never'}

# 4. Sarcasm JSONL 데이터셋 불러오기
# 한 줄에 하나의 JSON 객체가 저장된 형식
with open('Sarcasm.json', 'r', encoding='utf-8-sig') as f:

    # 5. 뉴스 데이터를 한 줄씩 순회
    for line in f:

        # 빈 줄은 건너뛰기
        if not line.strip():
            continue

        # JSON 문자열을 Python 딕셔너리로 변환
        item = json.loads(line)

        # 6. 뉴스 헤드라인, 레이블, URL 추출
        sentence = item['headline'].lower()
        label = int(item['is_sarcastic'])
        link = item['article_link']

        # 7. HTML 태그 제거
        # <br>, <p> 등의 태그를 제거하고 텍스트만 추출
        soup = BeautifulSoup(sentence, 'html.parser')
        sentence = soup.get_text(' ')

        # 8. 구두점 제거
        # 마침표, 쉼표, 느낌표 등을 공백으로 변환
        sentence = sentence.translate(table)

        # 9. 문장을 공백 기준으로 단어 분리
        words = sentence.split()

        # 10. 불용어 제거
        filtered_words = []

        for word in words:

            # 불용어가 아닌 단어만 추가
            if word not in stopword_set:
                filtered_words.append(word)

        # 11. 남은 단어를 하나의 문장으로 결합
        filtered_sentence = ' '.join(filtered_words)

        # 12. 전처리 결과 저장
        sentences.append(filtered_sentence)
        labels.append(label)
        urls.append(link)
```

```PYTHON
# 2. Tokenizer 설정
# 사용할 최대 어휘 크기
vocab_size = 23000

# 문장 최대 길이: 10개 토큰
max_length = 10

# 문장이 10개 토큰보다 길면 뒤쪽 토큰 제거
trunc_type = 'post'

# 문장이 10개 토큰보다 짧으면 뒤쪽에 0 추가
padding_type = 'post'

# 학습하지 않은 단어를 대체하는 OOV 토큰
oov_tok = '<OOV>'

# Tokenizer 객체 생성
# num_words=23000: 사용할 단어 인덱스를 23,000 미만으로 제한
# oov_token: 미등록 단어를 OOV 인덱스로 변환
tokenizer = Tokenizer(
    num_words=vocab_size,
    oov_token=oov_tok
)

# 3. 학습 데이터로 단어 사전 생성
# 학습 문장에서 단어별 등장 빈도를 계산하여 정수 인덱스 부여
# 테스트 데이터로는 Tokenizer를 학습하지 않음
tokenizer.fit_on_texts(training_sentences)

# 생성된 단어 사전 확인
word_index = tokenizer.word_index

print("전체 단어 사전 크기:", len(word_index))
print("단어 사전 일부:", list(word_index.items())[:20])


# 4. 학습 문장 정수 시퀀스 변환
# 단어를 정수 인덱스로 변환
# 학습 데이터에 없는 단어는 OOV 인덱스 1로 처리
training_sequences = tokenizer.texts_to_sequences(
    training_sentences
)

# 모든 문장 길이를 10으로 통일
# 짧은 문장은 뒤쪽에 0 추가

    padding=padding_type,
    truncating=trunc_type

# 테스트 데이터의 새로운 단어는 OOV로 처리
testing_sequences = tokenizer.texts_to_sequences(
    testing_sentences
)

# 학습 데이터와 동일한 방식으로 패딩 적용
testing_padded = pad_sequences(

    maxlen=max_length,
    padding=padding_type,
    truncating=trunc_type
)

# 6. 레이블을 NumPy 배열로 변환
# 이진 분류 정답: 0 또는 1
training_labels = np.array(training_labels)
testing_labels = np.array(testing_labels)

print("학습 데이터 크기:", training_padded.shape)
print("테스트 데이터 크기:", testing_padded.shape)

print("학습 레이블 크기:", training_labels.shape)
print("테스트 레이블 크기:", testing_labels.shape)

print("첫 번째 원본 문장:", training_sentences[0])
```
