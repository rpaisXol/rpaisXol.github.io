# 7.1 텐서플로 임베딩

- 신경망을 훈련하여 학습된 단어의 벡터 표현
- Dense와 Conv2D처럼 tf.keras는 층을 사용해 임베딩을 구현
- 이 층은 정수를 임베딩 벡터로 매핑하는 룩업 테이블을 만듬
- 이 테이블은 정수로 표현된 단어의 벡터를 담고 있다.
- 실제 자연어 처리 문제에서는 2차원보다 훨씬 더 많은 차원을 사용 
- 벡터 공간에 있는 한 벡터의 방향이 단어의 의미를 인코딩
- 즉, 거의 비슷한 방향을 가리키는 벡터를 가진 단어는 서로 연관

```python

import json
import string
from bs4 import BeautifulSoup

# 1. 문장, 레이블, URL을 저장할 리스트
sentences = []
labels = []
urls = []

# 2. 구두점 제거를 위한 변환 테이블
# string.punctuation: 일반적인 영어 구두점 목록
# translate(): 지정된 구두점 제거
table = str.maketrans('', '', string.punctuation)

# 3. 앞에서 정의한 불용어 리스트 사용
# set으로 변환하여 검색 속도 향상
stopword_set = set(word.lower() for word in stopwords)

# 4. Sarcasm 데이터셋 불러오기
# JSONL 형식: 한 줄에 하나의 JSON 객체 저장
# utf-8-sig: UTF-8 BOM이 있는 경우에도 처리
with open("Sarcasm.json", 'r', encoding='utf-8-sig') as f:

    # 5. JSON 파일을 한 줄씩 읽기
    for line in f:

        # 빈 줄 건너뛰기
        if not line.strip():
            continue

        # JSON 문자열을 Python 딕셔너리로 변환
        item = json.loads(line)

        # 6. 뉴스 헤드라인을 소문자로 변환
        sentence = item['headline'].lower()

        # 7. 구두점 앞뒤에 공백 추가
        sentence = sentence.replace(",", " , ")
        sentence = sentence.replace(".", " . ")
        sentence = sentence.replace("_", " _ ")
        sentence = sentence.replace("/", " / ")

        # 8. BeautifulSoup로 HTML 태그 제거
        soup = BeautifulSoup(sentence, 'html.parser')
        sentence = soup.get_text(' ')

        # 9. 공백을 기준으로 단어 분리
        words = sentence.split()

        # 10. 불용어 제거
        filtered_sentence = ""

        for word in words:

            # 구두점 제거
            word = word.translate(table)

            # 빈 문자열과 불용어 제외
            if word and word not in stopword_set:
                filtered_sentence += word + " "

        # 11. 전처리된 문장 저장
        sentences.append(filtered_sentence.strip())

        # 12. 풍자 여부 레이블 저장
        # 0: 일반 뉴스, 1: 풍자 뉴스
        labels.append(int(item['is_sarcastic']))

        # 13. 기사 URL 저장
        urls.append(item['article_link'])

# 14. 결과 확인
print("전체 문장 수:", len(sentences))
print("전체 레이블 수:", len(labels))
print("전체 URL 수:", len(urls))

print("첫 번째 문장:", sentences[0])
print("첫 번째 레이블:", labels[0])
print("첫 번째 URL:", urls[0])
```

```python
# 2) Tokenizer 및 패딩: 텍스트를 동일한 길의 정수 시퀸스로 변환

# 1. 하이퍼파라미터 설정
vocab_size = 10000       # 최대 어휘 크기
embedding_dim = 16       # 임베딩 벡터 차원
max_length = 10          # 최대 문장 길이
trunc_type = 'post'      # 긴 문장은 뒤쪽 토큰 제거
padding_type = 'post'    # 짧은 문장은 뒤쪽에 0 추가
oov_tok = '<OOV>'        # 미등록 단어를 대체할 토큰

# 2. Tokenizer 생성
tokenizer = Tokenizer(
    num_words=vocab_size,
    oov_token=oov_tok
)

# 3. 학습 문장으로 단어 사전 생성
tokenizer.fit_on_texts(training_sentences)

# 4. 학습 문장을 정수 시퀀스로 변환
training_sequences = tokenizer.texts_to_sequences(
    training_sentences
)

# 5. 학습 문장 패딩
training_padded = pad_sequences(
    training_sequences,
    maxlen=max_length,
    padding=padding_type,
    truncating=trunc_type
)

# 6. 테스트 문장을 정수 시퀀스로 변환
testing_sequences = tokenizer.texts_to_sequences(
    testing_sentences
)

# 7. 테스트 문장 패딩
testing_padded = pad_sequences(
    testing_sequences,
    maxlen=max_length,
    padding=padding_type,
    truncating=trunc_type
)

# 8. NumPy 배열 변환
training_padded = np.array(training_padded, dtype=np.int32)
training_labels = np.array(training_labels, dtype=np.int32)

testing_padded = np.array(testing_padded, dtype=np.int32)
testing_labels = np.array(testing_labels, dtype=np.int32)

# 9. 결과 확인
print("학습 데이터:", training_padded.shape)
print("테스트 데이터:", testing_padded.shape)
print("단어 사전 크기:", len(tokenizer.word_index))
```

```python
# 3) 신경망 정의: 임베딩 기반 이진 분류 모델 구성

# 1. 임베딩 층을 초기화하기 위해
# 어휘 사전 크기와 임베딩 차원을 먼저 지정
vocab_size = 10000
embedding_dim = 16

# 2. 신경망 모델 구성
model = Sequential([

    # 입력층: 정수로 변환된 10개의 토큰 입력
    Input(shape=(max_length,)),

    # Embedding: 각 단어를 16차원 벡터로 변환
    # mask_zero=True: 패딩값 0을 마스킹
    Embedding(
        input_dim=vocab_size,
        output_dim=embedding_dim,
        mask_zero=True
    ),

    # 단어 임베딩 벡터들의 평균 계산
    # 문장의 특징을 하나의 16차원 벡터로 압축
    GlobalAveragePooling1D(),

    # 은닉층: 뉴런 24개, ReLU 활성화 함수
    Dense(24, activation='relu'),

    # 출력층: 풍자 여부를 0~1 사이의 확률로 예측
    Dense(1, activation='sigmoid')
])

# 3. 모델 컴파일
model.compile(
    loss='binary_crossentropy',  # 이진 분류 손실 함수
    optimizer='adam',            # Adam 옵티마이저
    metrics=['accuracy']         # 분류 정확도
)

# 4. 모델 구조 확인
model.summary()
```

1. 임베딩 층을 초기화하기 위해서 어휘 사전 크기와 임베딩 차원을 먼저 지정
2. 각 단어에 대해 embedding_dim 차원의 배열을 초기화, 예를 들면 embedding_dim이 16이라면 어휘 사전에 있는 각 단어는 16차원 벡터에 할당
3. 훈련 데이터의 입력과 레이블을 매핑하기 위해 신경망 훈련될 때 역전파를 통해서 이 벡터가 학습함

- 임베딩 층의 어휘 사전 크기는 10,000이고 임베딩 차원(각 단어의 벡터)은 16
- 임베딩 층의 훈련 가능한 전체 파라미터 개수는 160,000개 이상
- 평균 풀링 층은 임베딩 층의 출력을 단순히 평균해 16차원의 벡터를 만들기 위해 훈련 가능한 파라미터가 0개
- 24개의 뉴런을 가진 밀집 층에 이를 전달하고, 밀집 층은 가중치와 절편을 가지고 있으므로 (24 X 16) + 16 = 408개의 파라미터를 학습
- 이 층의 출력이 하나의 뉴런을 가진 마지막 층에 전달되고, 이 층은 (1 X 24) + 1 = 25개의 파라미터를 가짐
- 모델 훈련 시 30번의 epochs 후 99% 이상 높은 정확도를 얻음, 그러나 검증 정확도는 81%

```python

# epochs=30: 최대 30회 학습
# batch_size=32: 한 번에 32개 데이터 처리

    validation_split=0.2,
```

## 7.1.2 언어 모델의 과대 적합 줄이기

- 신경망이 훈련 데이터에서 과도하게 맞춰질 경우는 과대적합 현상 발생
- 훈련 데이터의 다른 데이터보다 잡은 데이터에 있는 패턴에 잘 맞춰질 경우가 많이 발생
- 이런 특별한 잡음이 검증 세트에는 없기 때문에 신경망이 이를 잘 마출 수 있도록 검증 세트의 손실이 더 나빠짐
- 새로 학습 설정하기

```python

# 학습 설정
epochs = 100
batch_size = 32

# 결과 저장 폴더 생성
save_dir = 'training_results'
os.makedirs(save_dir, exist_ok=True)

# 모델 생성, 컴파일 및 학습 함수
def train_model(learning_rate, model_name):

    # 랜덤 시드 초기화
    reset_seed()

    # 새로운 신경망 모델 생성
    model = create_model()

    # 두 실험 모두 동일한 초기 가중치 적용
    model.set_weights(initial_weights)

    # Adam 옵티마이저 설정
    adam = Adam(
        learning_rate=learning_rate,
        beta_1=0.9,
        beta_2=0.999,
        amsgrad=False
    )

    # 모델 컴파일
    model.compile(
        loss='binary_crossentropy',
        optimizer=adam,
        metrics=['accuracy']
    )

    # 학습 전 초기 정확도 확인
    initial_loss, initial_acc = model.evaluate(
        training_padded,
        training_labels,
        verbose=0
    )

    print("\n모델 이름:", model_name)
    print("학습률:", model.optimizer.learning_rate.numpy())
    print("초기 손실:", initial_loss)
    print("초기 정확도:", initial_acc)

    # 학습 전 옵티마이저 업데이트 횟수 확인
    # 새로운 옵티마이저라면 0이어야 함
    print("업데이트 횟수:",
          model.optimizer.iterations.numpy())

    # 검증 손실이 가장 낮은 모델의 가중치 저장
    # 100 Epoch 학습은 유지하되 최적 가중치를 별도로 보관
    checkpoint = tf.keras.callbacks.ModelCheckpoint(
        filepath=os.path.join(
            save_dir, model_name + '.weights.h5'
        ),
        monitor='val_loss',
        mode='min',
        save_best_only=True,
        save_weights_only=True,
        verbose=0
    )

    # 모델 학습
    history = model.fit(
        training_padded,
        training_labels,
        epochs=epochs,
        batch_size=batch_size,
        validation_data=(
            validation_padded,
            validation_labels
        ),
        shuffle=True,
        callbacks=[checkpoint],
        verbose=1
    )

    return model, history

```
### 해결방법 1: 학습률 조정하기

- 과대적합을 일으키는 가장 큰 요소는 옵티마이저의 학습률이 너무 높은 경우
- 즉, 신경망이 너무 빠르게 학습한다는 의미
- 예를 들면 옵티마이저 함수를 기존 adam에서 매개 변수를 사용해 변경 가능

- amsgrad는 "On the Convergence of Adam and Beyond"에서 소개된 또 다른 Adam 옵티마이저 구현

### 해결방법 2: 어휘사전 크기 바꾸기

- Sarcasm 데이터셋에서는 단어를 다루기 때문에 데이터셋에 있는 단어, 특히 빈도를 살펴보면 과대적합 이슈를 고치는 데 도움이 되는 단서를 발견 가능
- tokenizer 객체의 word_count 속성을 사용하면 단어의 빈도를 확인 가능
- 이 속성을 출력하면 단어와 단어의 카운트의 튜플을 담은 OrderedDict가 반환
- 제공한 단어 빈도 그래프를 보면, 일부 단어는 매우 자주 등장하지만 대부분의 단어는 등장 빈도가 낮은 **롱테일(Long-tail) 분포**를 나타냅니다.
- 등장 빈도가 낮은 단어까지 모두 학습하면 모델이 학습 데이터에만 존재하는 단어의 특징을 과도하게 학습할 수 있습니다. 
- 따라서 어휘 사전 크기(`vocab_size`)를 줄여 과적합을 완화할 수 있는지 확인하겠습니다

**OOV 토큰 비율 확인**

어휘 사전 크기를 지나치게 줄이면 너무 많은 단어가 OOV로 변환되어 정보 손실이 발생할 수 있습니다.

```python
# OOV 토큰 비율 계산 함수
def calculate_oov_ratio(padded):

    # 패딩값 0을 제외한 전체 토큰 개수
    total_tokens = np.count_nonzero(padded)

    # OOV 토큰(인덱스 1) 개수
    oov_tokens = np.sum(padded == 1)

    return oov_tokens / total_tokens * 100 if total_tokens else 0.0


# 기존 어휘 사전과 변경된 어휘 사전 비교
print("기존 어휘 사전 (10000)")
print(f"학습 OOV 비율: {calculate_oov_ratio(training_padded):.2f}%")
print(f"검증 OOV 비율: {calculate_oov_ratio(validation_padded):.2f}%")

print("\n변경된 어휘 사전 (2000)")
print(f"학습 OOV 비율: {calculate_oov_ratio(training_padded_small):.2f}%")
print(f"검증 OOV 비율: {calculate_oov_ratio(validation_padded_small):.2f}%")
```

**임베딩 신경망 재정의 및 컴파일**

기존 모델의 가중치를 재사용하여 추가 학습하는 것이 아니라 새로운 모델을 생성합니다.

```python

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import (
    Input, Embedding, GlobalAveragePooling1D, Dense
)
from tensorflow.keras.optimizers import Adam

# 1. 랜덤 시드 설정
tf.keras.utils.set_random_seed(42)

# 2. 새로운 신경망 모델 정의
model_small = Sequential([

    # 입력층: 최대 10개의 정수 토큰
    Input(shape=(max_length,), dtype='int32'),

    # 임베딩층: 어휘 크기 2000, 임베딩 차원 16
    Embedding(vocab_size_small, embedding_dim),

    # 단어 임베딩 벡터들의 평균 계산
    GlobalAveragePooling1D(),

    # 은닉층: 뉴런 24개
    Dense(24, activation='relu'),

    # 출력층: 빈정거림 여부 예측
    Dense(1, activation='sigmoid')
])

# 3. 앞선 비교 실험에서 저장한 초기 가중치 적용
# 임베딩층은 앞의 2000개 행만 사용
# 나머지 Dense 층은 동일한 초기 가중치 사용
small_initial_weights = [
    initial_weights[0][:vocab_size_small].copy()
] + [w.copy() for w in initial_weights[1:]]

model_small.set_weights(small_initial_weights)

# 4. Adam 옵티마이저 정의
adam_small = Adam(
    learning_rate=0.0001,
    beta_1=0.9,
    beta_2=0.999,
    amsgrad=False
)

# 5. 모델 컴파일
model_small.compile(
    optimizer=adam_small,
    loss='binary_crossentropy',
    metrics=['accuracy']
)

# 6. 모델 구조 확인
model_small.summary()
```

```python
    mode='min',

    save_weights_only=True,
    verbose=0

history_small = model_small.fit(
    training_padded_small,
    training_labels,
    epochs=100,
    batch_size=32,
    validation_data=(
        validation_padded_small,
        validation_labels
    ),
    shuffle=True,
    callbacks=[checkpoint_small],
    verbose=1
)

```

```python
    # lower(): 모든 문자를 소문자로 변환

    # BeautifulSoup로 HTML 태그 제거
    # <br> 등의 HTML 태그를 제거하고 텍스트만 추출
    soup = BeautifulSoup(sentence, 'html.parser')

    # 공백을 기준으로 문장을 단어 단위로 분리
    words = sentence.split()
```

```python
        link = item['article_link']

        # 7. HTML 태그 제거
        # <br>, <p> 등의 태그를 제거하고 텍스트만 추출
        soup = BeautifulSoup(sentence, 'html.parser')
        sentence = soup.get_text(' ')

        # 8. 구두점 제거
        # 마침표, 쉼표, 느낌표 등을 공백으로 변환
        sentence = sentence.translate(table)

        # 9. 문장을 공백 기준으로 단어 분리
        words = sentence.split()

        # 10. 불용어 제거
        filtered_words = []

        for word in words:

            # 불용어가 아닌 단어만 추가
            if word not in stopword_set:
                filtered_words.append(word)

        # 11. 남은 단어를 하나의 문장으로 결합
        filtered_sentence = ' '.join(filtered_words)

        # 12. 전처리 결과 저장
        sentences.append(filtered_sentence)
        labels.append(label)
        urls.append(link)


# 13. Tokenizer 생성
# num_words=20000: 사용할 단어 인덱스 범위 제한
# oov_token: 학습되지 않은 단어를 대체할 토큰
vocab_size = 20000

tokenizer = Tokenizer(
    num_words=vocab_size,
    oov_token='<OOV>'
)

# 14. 전체 문장으로 단어 사전 생성
tokenizer.fit_on_texts(sentences)

# 15. 문장을 정수 시퀀스로 변환
sequences = tokenizer.texts_to_sequences(sentences)

# 16. 패딩 적용
# 모든 문장의 길이를 100개 토큰으로 통일
# 짧으면 0 추가, 길면 뒤쪽 토큰 제거
padded = pad_sequences(
    sequences,
    maxlen=100,
    padding='post',
    truncating='post'
)

# 17. 결과 확인
print("전체 문장 수:", len(sentences))
print("전체 레이블 수:", len(labels))
print("전체 URL 수:", len(urls))

print("단어 사전 크기:", len(tokenizer.word_index))
print("첫 번째 문장:", sentences[0])
print("첫 번째 레이블:", labels[0])
print("첫 번째 URL:", urls[0])
print("첫 번째 시퀀스:", sequences[0])
print("패딩 데이터 크기:", padded.shape)
```

```PYTHON
# 2. Tokenizer 설정
```
