# 5.1 데이터셋 시작하기

텐서플로 데이터셋이 설치되고 나면 tfds.load 함수에 원하는 데이터셋 이름을 전달해 데이터를 가져옴

```python

import tensorflow as tf

mnist_data = tfds.load("fashion_mnist")

for item in mnist_data: # 패션 데이터셋은 train, test을 키로 갖는 딕셔너리
	print(item)
```

결과로 PrefetchDataset의 객체가 출력, 이 객체를 순회하면서 데이터를 조사할 수 있다. 이 클래스의 좋은 기능 하나는 take(1) 메서드를 호출해 첫 번째 레코드를 가져올 수 있다는 점이다.

```python
# MNIST 학습 데이터셋에서 첫 번째 샘플 1개 추출
# take(1): 데이터셋에서 앞의 데이터 1개를 가져옴
for item in mnist_train.take(1):

    # 데이터의 전체 내용 출력
    # 일반적으로 이미지(image)와 정답 레이블(label)을 포함
    print(item)

    # 데이터에 포함된 키(Key) 확인
    # 각 데이터가 어떤 항목으로 구성되어 있는지 확인
    print(item.keys())
    
    # 이미지 데이터 추출
    print(item["image"].shape) # (28, 28, 1)
    
    # 실제 정답 레이블 추출
    print(item["label"].shape) # ()
```

- 첫 번째 print 문의 출력은 각 레코드 타입의 딕셔너리라는 것을 보여줌
- 이 딕셔너리의 키를 출력하면 image와 label

```python
# 데이터셋 로드 시 with_info 매개변수를 사용하면 데이터셋 정보 얻음
mnist_test, info = tfds.load(name="fashion_mnist", with_info="true")
```

# 5.3 데이터 증강을 위해 매핑 함수 사용

- 텐서플로 데이터셋을 사용할 때 이미지 증식을 수행하는 방법
- 가장 좋은 방법은 데이터셋 객체에 매핑 함수를 사용

```python
# 데이터셋 변환해 매핑하려면 매핑 함수 정의
def augmentimages(image, label):
    image = tf.cast(image, tf.float32)
    image = (image / 255.0)
    image = tf.image.random_flip_left_right(image) # 이미지를 랜덤하게 오른쪽이나 왼쪽으로 뒤집음
    return image, label


#데이터에 매핑해 새로운 데이터셋 train를 생성
train = data.map(augmentimages)

train_batches = train.shuffle(100).batch(10)
```

## 5.3.1 텐서플로 에드온 사용하기

- 텐서플로 애드온 라이브러리는 더 많은 함수 제공
- ImageDataGenerator에 있는 일부 기능은 에드온만 제공

```python
%pip install tensorflow-addons
```

다음 코드는 앞에서 만든 매핑 함수에 에드온 추가

```python
import tensorflow_addons as tfa

def augmentimages(image, label):
    image = tf.cast(image, tf.float32)
    image = (image / 255.0)
    image = tf.image.random_flip_left_right(image) # 이미지를 랜덤하게 오른쪽이나 왼쪽으로 뒤집음
    image = tfa.image.rotate(image, tf.random.uniform(shape=[], minval=-0.1, maxval=0.1)) # 이미지를 랜덤하게 회전
    return image, label
```

# 5.4 사용자 정의 분할 사용

- 텐서플로에선 원하는 형태로 데이터셋 분할을 할 수 있는 API가 존재한다.
- 예) 개와 고양이 분류 데이터셋

```python

import tensorflow_datasets as tfds

# 1. 전체 학습 데이터 불러오기
# split='train': train 분할에 포함된 모든 이미지 사용
# as_supervised=True: 데이터를 (image, label) 튜플 형태로 반환
# image: 고양이 또는 강아지 이미지
# label: 정답 클래스 (고양이=0, 강아지=1)
data = tfds.load('cats_vs_dogs', split='train', as_supervised=True)


# 2. 처음 10,000개 데이터만 불러오기
# train[:10000]: train 데이터의 인덱스 0~9999 선택
# 전체 데이터 중 앞에서부터 10,000개 사용
data = tfds.load('cats_vs_dogs', split='train[:10000]', as_supervised=True)


# 3. 전체 데이터의 처음 20%만 불러오기
# train[:20%]: train 데이터의 앞쪽 20% 선택
# 데이터 개수 대신 비율(%)을 기준으로 분할
data = tfds.load('cats_vs_dogs', split='train[:20%]', as_supervised=True)


# 4. 마지막 1,000개와 처음 1,000개 데이터 결합
# train[-1000:]: train 데이터의 마지막 1,000개 선택
# train[:1000]: train 데이터의 처음 1,000개 선택
# + : 선택한 두 데이터 구간을 순서대로 연결
# 결과: 총 2,000개 데이터
data = tfds.load('cats_vs_dogs', split='train[-1000:] + train[:1000]', as_supervised=True)
```

강아지-고양이 데이터셋은 훈련 세트, 검증 세트, 테스트 세트 분할이 없지만 텐서플로 데이터셋을 사용해 세트가 분할이 가능함.

예) 데이터셋을 80%, 10%, 10%으로 분할한다고 할 때
```python

import tensorflow_datasets as tfds

# 1. 학습 데이터 (Training Data)
# 전체 데이터셋의 처음 80%를 학습 데이터로 사용
# train[:80%]: 시작부터 전체의 80% 지점까지 선택
# as_supervised=True: (image, label) 튜플 형태로 반환
# 모델의 가중치와 편향을 학습하는 데 사용
train_data = tfds.load(
    'cats_vs_dogs',
    split='train[:80%]',
    as_supervised=True
)


# 2. 검증 데이터 (Validation Data)
# 전체 데이터셋의 80%~90% 구간을 검증 데이터로 사용
# train[80%:90%]: 80% 지점부터 90% 지점 전까지 선택
# 학습 중 모델 성능과 과적합 여부를 확인하는 데 사용
val_data = tfds.load(
    'cats_vs_dogs',
    split='train[80%:90%]',
    as_supervised=True
)


# 3. 테스트 데이터 (Test Data)
# 전체 데이터셋의 마지막 10%를 테스트 데이터로 사용
# train[90%:]: 90% 지점부터 마지막 데이터까지 선택
# 학습이 완료된 모델의 최종 성능 평가에 사용
test_data = tfds.load(
    'cats_vs_dogs',
    split='train[90%:]',
    as_supervised=True
)


# 4. 각 데이터셋의 샘플 개수 확인
print("학습 데이터:", train_data.cardinality().numpy())
print("검증 데이터:", val_data.cardinality().numpy())
print("테스트 데이터:", test_data.cardinality().numpy())
```

- 변환된 데이터셋 길이 정보를 제공하지 않기 때문에, 원본 세트를 올바르게 분할했는지 확인이 어려움
- 따라서 얼마나 많은 레코드가 분할 세트에 포함되었는지 확인하려면 전체 세트를 순회하며 하나씩 카운트 해야함

```python
# 방법1
train_length = [i for i, _ in enumerate(train_data)][-1] + 1
print(train_length)

# 방법2 - 이 방법을 추천
train_length = tf.data.experimental.cardinality(train_data).numpy()
print(train_length)
```

# 5.5 TFRecord 이해하기

- 텐서플로 데이터셋은 데이터를 다운로드해 디스크에 캐싱
- 사용할 때마다 데이터를 다운로드하지 않음
- 텐서플로 데이터셋은 캐싱을 위해 TFRecord 포맷을 사용함
- 텐서플로는 대용량의 데이터를 저장하고 추출할 때 선호하는 포맷
- 파일구조 매우 단순, 순차적으로 데이터를 읽음
- 디스크게 저장된 파일에서 각각 레코드는 레코드 길이 정수, 이 정수의 순환 중복 검사, 데이터의 대규모 바이트 배열, 바이트 배열의 CRC(순환 중복 검사)로 구성
- 레코드가 연결되어 파일을 구성하고 대규모 데이터셋은 샤딩(Sharding)

```python

import os
import tensorflow as tf
import tensorflow_datasets as tfds

# 1. MNIST 테스트 데이터셋 불러오기
# with_info=True: 데이터셋의 저장 경로와 메타데이터 반환
_, info = tfds.load('mnist', split='test', with_info=True)

# 2. 실제 TFRecord 파일 경로 확인
# 데이터셋 저장 폴더와 테스트 파일 이름 결합
filename = os.path.join(
    str(info.data_dir),
    info.splits['test'].filenames[0]
)

print("TFRecord 파일 경로:", filename)

# 3. TFRecord 파일 읽기
# TFRecordDataset: 바이너리 형식의 TFRecord 파일을 읽는 객체
raw_dataset = tf.data.TFRecordDataset(filename)

# 4. 첫 번째 레코드 확인
for raw_record in raw_dataset.take(1):

    # 원본 데이터의 바이트 크기 확인
    print("레코드 크기:", len(raw_record.numpy()))

    # 직렬화된 데이터를 tf.train.Example로 변환
    example = tf.train.Example()
    example.ParseFromString(raw_record.numpy())

    # 데이터에 포함된 필드 이름 확인
    print("데이터 필드:", list(example.features.feature.keys()))

    # 정답 레이블 추출
    label = example.features.feature['label'].int64_list.value[0]
    print("정답 레이블:", label)

    # 이미지 바이너리 데이터 추출
    image_bytes = example.features.feature['image'].bytes_list.value[0]

    # PNG 바이너리를 이미지 텐서로 변환
    image = tf.io.decode_png(image_bytes, channels=1)

    print("이미지 크기:", image.shape)
```

# 5.6 텐서플로에서 데이터 관리를 위한 ETL 프로세스

- ETL은 규모와 상관없이 텐서플로 훈련에  사용되는 핵심 패턴
- 작은 규모로 한 대의 컴퓨터에서 모델 구축 가능
- 여러 대의 머신에서 대용량 데이터셋으로 수행하는 대규모 훈련에서도 동일한 기술 사용 가능
- Extract(추출)는 다른곳에 저장된 원시 데이터를 가져와 반환위한 준비 단계
- Transform(변환)은 훈련에 적합하도록 또는 훈련이 잘되도록 데이터를 조작 단계
- 예) 이미지 증강, 특성 열 매핑, 데이터에 적용하는 변환 로직
- Load(적재)는 훈련을 위해 신경망에 데이터 로드 단계

말-사람 데이터셋으로 분류기를 훈련하는 전체 코드

```python
import tensorflow as tf
import tensorflow_datasets as tfds
import tensorflow_addons as tfa

from tensorflow.keras import Sequential
from tensorflow.keras.layers import Layer, Conv2D, MaxPooling2D, Flatten, Dense

# 모델 정의
model = Sequential([
    Conv2D(16, (3, 3), activation='relu', input_shape=(300, 300, 3)),
    MaxPooling2D((2, 2)),
    Conv2D(32, (3, 3), activation='relu'),
    MaxPooling2D((2, 2)),
    Conv2D(64, (3, 3), activation='relu'),
    MaxPooling2D((2, 2)),
    Conv2D(64, (3, 3), activation='relu'),
    MaxPooling2D((2, 2)),
    Conv2D(64, (3, 3), activation='relu'),
    MaxPooling2D((2, 2)),
    Flatten(),
    Dense(512, activation='relu'),
    Dense(1, activation='sigmoid')
])

model.compile(optimizer='Adam', loss='binary_crossentropy', metrics=['accuracy'])


# 추출 단계 시작
train_data = tfds.load('horses_or_humans', split='train', as_supervised=True)
val_data = tfds.load('horses_or_humans', split='test', as_supervised=True)

# 변환 단계 시작
def augmentimages(image, label):
    image = tf.cast(image, tf.float32)
    image = (image/255)
    image = tf.image.random_flip_left_right(image)
    image = tfa.image.rotate(image, 40, interpolation='NEAREST')
    return image, label

train = train_data.map(augmentimages)
train_batches = train.shuffle(100).batch(32)
validation_batches = val_data.batch(32)

# 로드 단계 시작
history = model.fit(train_batches, epochs=10, validation_data=validation_batches)
```

- 프로세스를 사용하면 데이터 파이프라인이나 데이터나 모델 구조의 변화에 덜 민감
- 텐서플로 데이터셋을 이용해 데이터를 추출하는 구조는 메모리에 들어갈 만큼 작은 데이터이거나 한 대의 머신에 담을 수 없는 큰 데이터이거나 데이터의 크기와 상관없이 동일
- 변환 된 후 데이터 로딩 단계도 하나의 CPU, GPU 클러스터 또는 TPU 포드에서 훈련하는지와는 상관없이 동일
- 데이터를 로드하는 단계가 훈련 속도에 큰 영향을 미침

## 5.6.1 로드 단계 최적화

- 데이터 추출과 변환은 CPU를 포함해 어느 프로세서에도 수행 가능
- GPU나 TPU는 데이터 다운로드, 해제, 레코드 순회 처리와 같은 단계에서 사용하는 코드를 위해서 만들어진 것은 아니다.
- 로드 코드는 CPU에서 실행, 훈련할 때는 GPU나 TPU로부터 큰 도움을 받는다.
- GPU와 TPU가 준비되어 있다면 CPU와 GPU/TPU 사이에 작업 부하를 잘 분배
- 추출과 변환은 CPU에서 수행되고 로드 단계는 GPU/TPU에서 수행

1) 첫 번째 배치가 준비되는 동안 GPU / TPU는 유휴 상태
2) 배치가 준비되면 훈련을 위해 GPU / TPU에 전송
3) 훈련이 끝난 후 두 번째 배치를 준비하기 시작할 때 까지 GPU는 유휴 상태
4) 유휴 시간이 많으므로 최적화할 여지가 있음

![[5) 텐서플롯 데이터셋으로 공개 데이터셋 사용-1794009631619.webp]]

- 이렇게 데이터 준비와 훈련을 병렬로 수행하는 것이 논리적인 해결책
- 이런 과정을 파이프라이닝(Pipelining)

![[5) 텐서플롯 데이터셋으로 공개 데이터셋 사용-1794010397527.webp]]

1)  이런 경우 CPU가 첫 번째 배치를 준비하는 동안 GPU/TPU가 아무런 일을 하지 않기 때문에 유휴 상태
2) 첫 번째 배치가 완료되면 GPU/TPU가 훈련을 시작
3) 물론 N-1번째 배치(CPU 훈련시간)를 훈련하는 시간과 N번째 배치(GPU 훈련시간)를 준비하는 시간이 항상 같지는 않음
4) 훈련 시간이 더 빠르면 GPU/TPU에 유휴 시간이 생김
5) 훈련 시간이 더 느리면 CPU에 유휴 시간이 생힘
6) 결국엔 적절한 배치 크기를 정하면 이 간격을 최적화하는 데 도움
7) GPU/TPU 비용이 비싸므로 가능하면 GPU/TPU의 유휴 시간을 줄이는게 좋다.
8) 그래서 코드 상에서 배치 사이즈를 잘 조절하는 것이 중요함

## 5.6.2 훈련 속도 향상을 위한 ETL 병렬화

텐서플로는 추출과 변환 과정을 병렬화하는 데 필요한 API 제공

예) 강아지-고양이 데이터셋과 TFRecord 파일 목록 생성

```python
import os
import tensorflow as tf

# 1. Cats vs Dogs 4.0.0 데이터셋 경로 설정
# 사용자 홈 디렉터리의 tensorflow_datasets 폴더 사용
data_dir = os.path.expanduser(

# 4. 파일 목록 출력
# 각 파일 경로는 tf.string 텐서 형태로 반환
# decode('utf-8'): 파일 경로를 문자열로 변환
for filename in file_list:
    print(filename.numpy().decode('utf-8'))
```

```python
raw_dataset = file_list.interleave(
```

1) tf.Data.Dataset.list_files를 사용해 로드할 파일 목록을 만듬
2) 파일 목록을 얻으면 files.interleave를 사용해 데이터셋 로드
3) cycle_length 매개변수는 동시에 전처리할 입력 원소 개수를 지정
4) 디스크에서 레코드를 로드할 때 디코딩하는 매핑 함수는 cycle_length를 4로 지정했기에 이 프로세스는 동시에 4개의 레코드를 처리
5) 이 값을 지정하지 않으면 로컬 PC에서 사용 가능한 CPU 코어 개수 사용
6) num_parallel_calls 매개 변수는
