# 4.1 데이터셋 시작하기

텐서플로 데이터셋이 설치되고 나면 tfds.load 함수에 원하는 데이터셋 이름을 전달해 데이터를 가져옴

```python

import tensorflow as tf
import tensorflow_datasets as tfds

mnist_data = tfds.load("fashion_mnist")

for item in mnist_data: # 패션 데이터셋은 train, test을 키로 갖는 딕셔너리
	print(item)
```

결과로 PrefetchDataset의 객체가 출력, 이 객체를 순회하면서 데이터를 조사할 수 있다. 이 클래스의 좋은 기능 하나는 take(1) 메서드를 호출해 첫 번째 레코드를 가져올 수 있다는 점이다.

```python
# MNIST 학습 데이터셋에서 첫 번째 샘플 1개 추출
# take(1): 데이터셋에서 앞의 데이터 1개를 가져옴
for item in mnist_train.take(1):

    # 데이터의 전체 내용 출력
    # 일반적으로 이미지(image)와 정답 레이블(label)을 포함
    print(item)

    # 데이터에 포함된 키(Key) 확인
    # 각 데이터가 어떤 항목으로 구성되어 있는지 확인
    print(item.keys())
    
    # 이미지 데이터 추출
    print(item["image"].shape) # (28, 28, 1)
    
    # 실제 정답 레이블 추출
    print(item["label"].shape) # ()
```

- 첫 번째 print 문의 출력은 각 레코드 타입의 딕셔너리라는 것을 보여줌
- 이 딕셔너리의 키를 출력하면 image와 label

```python
# 데이터셋 로드 시 with_info 매개변수를 사용하면 데이터셋 정보 얻음
mnist_test, info = tfds.load(name="fashion_mnist", with_info="true")
```

- 케라스 데이터셋은 ndarray 타입

```python
(training_images, training_labels), (test_images, test_labels) = tfds.as_numpy(tfds.load('fashion_mnist', split=['train', 'test'],   batch_size=-1, as_supervised=True))
```

# 4.3 데이터 증강을 위해 매핑 함수 사용

- 텐서플로 데이터셋을 사용할 때 이미지 증식을 수행하는 방법
- 가장 좋은 방법은 데이터셋 객체에 매핑 함수를 사용

```python
# 데이터셋 변환해 매핑하려면 매핑 함수 정의
def augmentimages(image, label):
    image = tf.cast(image, tf.float32)
    image = (image / 255.0)
    image = tf.image.random_flip_left_right(image) # 이미지를 랜덤하게 오른쪽이나 왼쪽으로 뒤집음
    return image, label


#데이터에 매핑해 새로운 데이터셋 train를 생성
train = data.map(augmentimages)

train_batches = train.shuffle(100).batch(10)
```

## 4.3.1 텐서플로 에드온 사용하기

- 텐서플로 애드온 라이브러리는 더 많은 함수 제공
- ImageDataGenerator에 있는 일부 기능은 에드온만 제공

```python
%pip install tensorflow-addons
```

다음 코드는 앞에서 만든 매핑 함수에 에드온 추가

```python
import tensorflow_addons as tfa

def augmentimages(image, label):
    image = tf.cast(image, tf.float32)
    image = (image / 255.0)
    image = tf.image.random_flip_left_right(image) # 이미지를 랜덤하게 오른쪽이나 왼쪽으로 뒤집음
    image = tfa.image.rotate(image, tf.random.uniform(shape=[], minval=-0.1, maxval=0.1)) # 이미지를 랜덤하게 회전
    return image, label
```

# 4.4 사용자 정의 분할 사용

- 텐서플로에선 원하는 형태로 데이터셋 분할을 할 수 있는 API가 존재한다.
- 예) 개와 고양이 분류 데이터셋

```python

import tensorflow_datasets as tfds

# 1. 전체 학습 데이터 불러오기
# split='train': train 분할에 포함된 모든 이미지 사용
# as_supervised=True: 데이터를 (image, label) 튜플 형태로 반환
# image: 고양이 또는 강아지 이미지
# label: 정답 클래스 (고양이=0, 강아지=1)
data = tfds.load('cats_vs_dogs', split='train', as_supervised=True)


# 2. 처음 10,000개 데이터만 불러오기
# train[:10000]: train 데이터의 인덱스 0~9999 선택
# 전체 데이터 중 앞에서부터 10,000개 사용
data = tfds.load('cats_vs_dogs', split='train[:10000]', as_supervised=True)


# 3. 전체 데이터의 처음 20%만 불러오기
# train[:20%]: train 데이터의 앞쪽 20% 선택
# 데이터 개수 대신 비율(%)을 기준으로 분할
data = tfds.load('cats_vs_dogs', split='train[:20%]', as_supervised=True)


# 4. 마지막 1,000개와 처음 1,000개 데이터 결합
# train[-1000:]: train 데이터의 마지막 1,000개 선택
# train[:1000]: train 데이터의 처음 1,000개 선택
# + : 선택한 두 데이터 구간을 순서대로 연결
# 결과: 총 2,000개 데이터
data = tfds.load('cats_vs_dogs', split='train[-1000:] + train[:1000]', as_supervised=True)
```

강아지-고양이 데이터셋은 훈련 세트, 검증 세트, 테스트 세트 분할이 없지만 텐서플로 데이터셋을 사용해 세트가 분할이 가능함.

예) 데이터셋을 80%, 10%, 10%으로 분할한다고 할 때
```python

import tensorflow_datasets as tfds

# 1. 학습 데이터 (Training Data)
# 전체 데이터셋의 처음 80%를 학습 데이터로 사용
# train[:80%]: 시작부터 전체의 80% 지점까지 선택
# as_supervised=True: (image, label) 튜플 형태로 반환
# 모델의 가중치와 편향을 학습하는 데 사용
train_data = tfds.load(
    'cats_vs_dogs',
    split='train[:80%]',
    as_supervised=True
)


# 2. 검증 데이터 (Validation Data)
# 전체 데이터셋의 80%~90% 구간을 검증 데이터로 사용
# train[80%:90%]: 80% 지점부터 90% 지점 전까지 선택
# 학습 중 모델 성능과 과적합 여부를 확인하는 데 사용
val_data = tfds.load(
    'cats_vs_dogs',
    split='train[80%:90%]',
    as_supervised=True
)


# 3. 테스트 데이터 (Test Data)
# 전체 데이터셋의 마지막 10%를 테스트 데이터로 사용
# train[90%:]: 90% 지점부터 마지막 데이터까지 선택
# 학습이 완료된 모델의 최종 성능 평가에 사용
test_data = tfds.load(
    'cats_vs_dogs',
    split='train[90%:]',
    as_supervised=True
)


# 4. 각 데이터셋의 샘플 개수 확인
print("학습 데이터:", train_data.cardinality().numpy())
print("검증 데이터:", val_data.cardinality().numpy())
print("테스트 데이터:", test_data.cardinality().numpy())
```

- 변환된 데이터셋 길이 정보를 제공하지 않기 때문에, 원본 세트를 올바르게 분할했는지 확인이 어려움
- 따라서 얼마나 많은 레코드가 분할 세트에 포함되었는지 확인하려면 전체 세트를 순회하며 하나씩 카운트 해야함

```python
# 방법1
train_length = [i for i, _ in enumerate(train_data)][-1] + 1
print(train_length)

# 방법2 - 이 방법을 추천
train_length = tf.data.experimental.cardinality(train_data).numpy()
print(train_length)
```

# 4.5 TFRecord 이해하기

- 텐서플로 데이터셋은 데이터를 다운로드해 디스크에 캐싱
- 사용할 때마다 데이터를 다운로드하지 않음
- 텐서플로 데이터셋은 캐싱을 위해 TFRecord 포맷을 사용함
- 텐서플로는 대용량의 데이터를 저장하고 추출할 때 선호하는 포맷
- 파일구조 매우 단순, 순차적으로 데이터를 읽음
